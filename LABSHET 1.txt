#include <stdio.h>

#define MAX 100

void readArray(int a[], int *n, const char *name) {
    int i;
    printf("Enter number of elements in %s (1-%d): ", name, MAX);
    scanf("%d", n);

    if (*n < 1 || *n > MAX) {
        printf("Invalid size. Please run the program again.\n");
        *n = 0;
        return;
    }

    printf("Enter %d elements: ", *n);
    for (i = 0; i < *n; i++) {
        scanf("%d", &a[i]);
    }
}

void displayArray(const int a[], int n) {
    int i;
    for (i = 0; i < n; i++) {
        printf("%d", a[i]);
        if (i < n - 1) printf(" ");
    }
    printf("\n");
}

/* 1. Sum of odd and even elements */
void sumOddEven(void) {
    int a[MAX], n, i, even = 0, odd = 0;
    readArray(a, &n, "the array");
    if (n == 0) return;

    for (i = 0; i < n; i++) {
        if (a[i] % 2 == 0) even += a[i];
        else odd += a[i];
    }

    printf("SumEven = %d\nSumOdd = %d\n", even, odd);
}

/* 2. Sum of two arrays of unequal sizes; missing values are treated as 0 */
void sumUnequalArrays(void) {
    int a[MAX], b[MAX], c[MAX * 2];
    int n, m, i, size;

    readArray(a, &n, "Array A");
    if (n == 0) return;
    readArray(b, &m, "Array B");
    if (m == 0) return;

    size = (n > m) ? n : m;
    for (i = 0; i < size; i++) {
        int x = (i < n) ? a[i] : 0;
        int y = (i < m) ? b[i] : 0;
        c[i] = x + y;
    }

    printf("Sum array: ");
    displayArray(c, size);
}

/* 3. Reverse an array */
void reverseArray(void) {
    int a[MAX], n, i;
    readArray(a, &n, "the array");
    if (n == 0) return;

    printf("Reversed array: ");
    for (i = n - 1; i >= 0; i--) {
        printf("%d", a[i]);
        if (i > 0) printf(" ");
    }
    printf("\n");
}

/* 4. Find kth smallest and kth largest element */
void kthSmallestLargest(void) {
    int a[MAX], n, i, j, temp, k;
    readArray(a, &n, "the array");
    if (n == 0) return;

    printf("Enter k (1-%d): ", n);
    scanf("%d", &k);
    if (k < 1 || k > n) {
        printf("Invalid k.\n");
        return;
    }

    /* Bubble sort in ascending order */
    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("%dth smallest = %d\n", k, a[k - 1]);
    printf("%dth largest = %d\n", k, a[n - k]);
}

/* 5. Union and intersection of two arrays (unique values) */
void unionIntersection(void) {
    int a[MAX], b[MAX], uni[MAX * 2], inter[MAX];
    int n, m, i, j, found, u = 0, in = 0;

    readArray(a, &n, "Array A");
    if (n == 0) return;
    readArray(b, &m, "Array B");
    if (m == 0) return;

    for (i = 0; i < n; i++) {
        found = 0;
        for (j = 0; j < u; j++)
            if (uni[j] == a[i]) found = 1;
        if (!found) uni[u++] = a[i];
    }

    for (i = 0; i < m; i++) {
        found = 0;
        for (j = 0; j < u; j++)
            if (uni[j] == b[i]) found = 1;
        if (!found) uni[u++] = b[i];
    }

    for (i = 0; i < n; i++) {
        int inB = 0, alreadyAdded = 0;
        for (j = 0; j < m; j++)
            if (a[i] == b[j]) inB = 1;
        for (j = 0; j < in; j++)
            if (inter[j] == a[i]) alreadyAdded = 1;
        if (inB && !alreadyAdded) inter[in++] = a[i];
    }

    printf("Union: ");
    displayArray(uni, u);
    printf("Intersection: ");
    if (in == 0) printf("None\n");
    else displayArray(inter, in);
}

/* 6. Find duplicate elements (each duplicate printed once) */
void findDuplicates(void) {
    int a[MAX], n, i, j, k, duplicate;
    readArray(a, &n, "the array");
    if (n == 0) return;

    printf("Duplicate elements: ");
    duplicate = 0;
    for (i = 0; i < n; i++) {
        int seenBefore = 0;
        for (j = 0; j < i; j++)
            if (a[i] == a[j]) seenBefore = 1;

        if (!seenBefore) {
            for (k = i + 1; k < n; k++) {
                if (a[i] == a[k]) {
                    printf("%d ", a[i]);
                    duplicate = 1;
                    break;
                }
            }
        }
    }
    if (!duplicate) printf("None");
    printf("\n");
}

/* 7. Check whether Array B is a subset of Array A */
void checkSubset(void) {
    int a[MAX], b[MAX], n, m, i, j, found, isSubset = 1;

    readArray(a, &n, "Array A");
    if (n == 0) return;
    readArray(b, &m, "Array B");
    if (m == 0) return;

    for (i = 0; i < m; i++) {
        found = 0;
        for (j = 0; j < n; j++) {
            if (b[i] == a[j]) {
                found = 1;
                break;
            }
        }
        if (!found) {
            isSubset = 0;
            break;
        }
    }

    printf("%s\n", isSubset ? "Yes, B is a subset of A."
                            : "No, B is not a subset of A.");
}

/* 8. Find the first repeating element by first occurrence */
void firstRepeating(void) {
    int a[MAX], n, i, j, found = 0;
    readArray(a, &n, "the array");
    if (n == 0) return;

    for (i = 0; i < n && !found; i++) {
        for (j = i + 1; j < n; j++) {
            if (a[i] == a[j]) {
                printf("First repeating element = %d\n", a[i]);
                found = 1;
                break;
            }
        }
    }

    if (!found) printf("No repeating element found.\n");
}

    int choice;

    do {
        printf("\n========== LABSHEET 1: ARRAY OPERATIONS ==========\n");
        printf("1. Sum of odd and even elements\n");
        printf("2. Sum of two unequal-sized arrays\n");
        printf("3. Reverse an array\n");
        printf("4. Find kth smallest and kth largest\n");
        printf("5. Find union and intersection\n");
        printf("6. Find duplicate elements\n");
        printf("7. Check whether one array is a subset of another\n");
        printf("8. Find the first repeating element\n");
        printf("0. Exit\n");
        printf("Enter your choice: ");

        if (scanf("%d", &choice) != 1) {
            printf("Invalid input. Please enter a number.\n");
            return 1;
        }

        switch (choice) {
            case 1: sumOddEven(); break;
            case 2: sumUnequalArrays(); break;
            case 3: reverseArray(); break;
            case 4: kthSmallestLargest(); break;
            case 5: unionIntersection(); break;
            case 6: findDuplicates(); break;
            case 7: checkSubset(); break;
            case 8: firstRepeating(); break;
            case 0: printf("Program terminated.\n"); break;
            default: printf("Invalid choice. Try again.\n");
        }
    } while (choice != 0);

    return 0;
}
